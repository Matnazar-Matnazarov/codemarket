from django import forms
from django.forms import inlineformset_factory
from ..models.model_project import Project
from ..models.model_project_image import ProjectImage
from ..models.model_database import ProjectBase
from ..models.model_language import ProjectLanguage
from django import forms

# Asosiy form
class ProjectForm(forms.ModelForm):
    class Meta:
        model = Project
        fields = [
            "name",
            "title",
            "about",
            "price",
            "url",
            "zip_file",
            # "technology",
        ]

class ProjectImageForm(forms.ModelForm):    
    class Meta:
        model = ProjectImage
        fields = ["name", "image"]

class ProjectLanguageForm(forms.ModelForm):    
    """
    Form for creating and editing project languages.
    
    This form is used to validate and process user input for project languages.
    It uses Django's built-in ModelForm to automatically generate form fields
    based on the ProjectLanguage model.
    """
    class Meta:
        """
        Metadata for the ProjectLanguageForm.
        
        This class defines the model and fields used by the form.
        """
        model = ProjectLanguage
        fields = ["technology", "language"]
